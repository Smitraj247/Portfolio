// Token Bucket Rate Limiter with Redis sliding window window algorithm
import { Redis } from '@upstash/redis';

export interface RateLimitResult {
  success: boolean;
  remainingTokens: number;
  resetTimeMs: number;
}

export async function checkRateLimit(
  userId: string, 
  limit = 100, 
  windowMs = 60000
): Promise<RateLimitResult> {
  const now = Date.now();
  const key = `ratelimit:${userId}:${Math.floor(now / windowMs)}`;
  
  const currentRequests = await Redis.incr(key);
  if (currentRequests === 1) {
    await Redis.pexpire(key, windowMs);
  }

  return {
    success: currentRequests <= limit,
    remainingTokens: Math.max(0, limit - currentRequests),
    resetTimeMs: (Math.floor(now / windowMs) + 1) * windowMs - now
  };
}
// High-performance in-memory LRU cache decorator with stale-while-revalidate support
export class SmartCache<T> {
  private store = new Map<string, { value: T; expiresAt: number }>();

  constructor(private defaultTTL = 300000) {} // 5 mins

  async getOrFetch(key: string, fetcher: () => Promise<T>): Promise<T> {
    const cached = this.store.get(key);
    if (cached && Date.now() < cached.expiresAt) {
      return cached.value;
    }

    const freshData = await fetcher();
    this.store.set(key, { value: freshData, expiresAt: Date.now() + this.defaultTTL });
    return freshData;
  }
}
// Custom React hook for debouncing search query inputs with memory cleanup
import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delayMs: number = 300): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebouncedValue(value), delayMs);
    return () => clearTimeout(timer);
  }, [value, delayMs]);

  return debouncedValue;
}
